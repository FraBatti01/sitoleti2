    function loadSavedArchive() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        return raw ? JSON.parse(raw) : [];

        return [];
      }
    }

    function persistSavedArchive(list) {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
      } catch (_) {}
      updateSavedBadge();
    }

    let savedArchive = loadSavedArchive();

    function updateSavedBadge() {
      document.getElementById("savedBadgeCount").textContent = savedArchive.length;
    }

    function isCardSaved(card) {
      if (!card) return false;
      return savedArchive.some(c => c.term.toLowerCase() === card.term.toLowerCase());
    }

    function toggleSaveCard(card) {
      if (!card) return;
      const idx = savedArchive.findIndex(c => c.term.toLowerCase() === card.term.toLowerCase());
      if (idx >= 0) {
        savedArchive.splice(idx, 1);
        persistSavedArchive(savedArchive);
        showToast("Rimossa dall'Archivio");
        if (state.mode === "saved") {
          loadDeck();
          return;
        }
      } else {
        savedArchive.unshift({ ...card, savedAt: Date.now() });
        persistSavedArchive(savedArchive);
        showToast("Salvata nell'Archivio ⭐");
      }
      updateBookmarkButtons();
    }

    function updateBookmarkButtons() {
      const card = state.deck[state.index];
      const saved = isCardSaved(card);
      const bFront = document.getElementById("bookmarkFrontBtn");
      const bBack = document.getElementById("bookmarkBackBtn");
      [bFront, bBack].forEach(btn => {
        btn.textContent = saved ? "★" : "☆";
        btn.classList.toggle("saved", saved);
      });
    }

    async function fetchAllVocabulary(count = 22) {
      // Pick 3 random seeds across all domains to generate a rich, mixed stream
      const pickedSeeds = shuffle(ALL_VOCAB_SEEDS).slice(0, 3);
      try {
        const dmResponses = await Promise.all(
          pickedSeeds.map(s => fetch(`https://api.datamuse.com/words?ml=${encodeURIComponent(s)}&max=20`).then(r => r.json()).catch(() => []))
        );
        const mergedWords = dmResponses.flat()
          .filter(w => /^[a-z]{6,15}$/i.test(w.word))
          .map(w => w.word);

        const pool = shuffle(Array.from(new Set([...pickedSeeds, ...mergedWords]))).slice(0, count + 8);
        const cards = [];

        await Promise.all(pool.map(async (word) => {
          if (cards.length >= count) return;
          try {
            const dictRes = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`);
            if (!dictRes.ok) return;
            const dictData = await dictRes.json();
            const entry = dictData[0];
            if (!entry || !entry.meanings || !entry.meanings.length) return;

            const meaning = entry.meanings[0];
            const defObj = meaning.definitions.find(d => d.example) || meaning.definitions[0];
            if (!defObj) return;

            const phonetic = entry.phonetic || (entry.phonetics && entry.phonetics.find(p => p.text)?.text) || "";
            const pos = meaning.partOfSpeech || "vocabulary";

            cards.push({
              cat: "Vocabulary",
              level: "C1",
              reg: "Formal / Academic",
              term: capitalize(entry.word),
              phonetic: phonetic,
              prompt: `${capitalize(pos)} • C1 Vocabulary`,
              it: `Lemma C1 (${pos}). Consulta definizione ed esempio per cogliere la sfumatura esatta.`,
              def: defObj.definition,
              ex: defObj.example || `In advanced English, "${entry.word}" is frequently used to express nuance.`
            });
          } catch (_) {}
        }));

        return cards.length > 0 ? shuffle(cards) : shuffle(FALLBACK_VOCAB_CARDS);
      } catch (e) {
        return shuffle(FALLBACK_VOCAB_CARDS);
      }
    }

    function shuffle(arr) {
      const a = [...arr];
      for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [a[i], a[j]] = [a[j], a[i]];
      }
      return a;
    }

      return str ? str.charAt(0).toUpperCase() + str.slice(1) : "";
    }

        let combined = shuffle([...liveCards, ...gramCards, ...vaultCards]);
        if (state.subFilter === "IDIOM") combined = vaultCards;
        if (state.subFilter === "GRAMMAR") combined = gramCards;

        state.deck = combined;

    function flipCard() {
      state.flipped = !state.flipped;

    function nextCard() {
      if (!state.deck.length) return;
      state.index = (state.index + 1) % state.deck.length;
      renderCard();
    }

    function prevCard() {
      if (!state.deck.length) return;
      state.index = (state.index - 1 + state.deck.length) % state.deck.length;
      renderCard();
    }

    function rateCard(rating) {
      const current = state.deck[state.index];
      if (!current) return;

      if (rating === "good") {
        state.mastered.add(current.term);
      } else if (rating === "again") {
        state.mastered.delete(current.term);
        const reinsertAt = Math.min(state.index + 3, state.deck.length);
        state.deck.splice(reinsertAt, 0, current);
      } else if (rating === "hard") {
        const reinsertAt = Math.min(state.index + 6, state.deck.length);
        state.deck.splice(reinsertAt, 0, current);
      }
      nextCard();
    }

    function speakText(text) {
      if (!("speechSynthesis" in window)) return;

      const clean = text.replace(/______/g, "blank");
      const utter = new SpeechSynthesisUtterance(clean);
      utter.lang = state.accent;
      utter.rate = 0.94;
      window.speechSynthesis.speak(utter);
    }

      state.accent = state.accent === "en-GB" ? "en-US" : "en-GB";
      e.target.textContent = state.accent === "en-GB" ? "🇬🇧 UK" : "🇺🇸 US";
    });

    document.getElementById("bookmarkFrontBtn").addEventListener("click", (e) => {
      e.stopPropagation();
      toggleSaveCard(state.deck[state.index]);
    });

    document.getElementById("bookmarkBackBtn").addEventListener("click", (e) => {
      e.stopPropagation();
      toggleSaveCard(state.deck[state.index]);
    });

    document.getElementById("speakFrontBtn").addEventListener("click", (e) => {
      e.stopPropagation();
      const card = state.deck[state.index];
      if (card) speakText(card.term);
    });

    document.getElementById("speakBackBtn").addEventListener("click", (e) => {
      e.stopPropagation();
      const card = state.deck[state.index];
      if (card) speakText(card.ex);
    });

    document.getElementById("recallBox").addEventListener("click", e => e.stopPropagation());
    document.getElementById("checkRecallBtn").addEventListener("click", () => {
      const card = state.deck[state.index];
      if (!card || !card.answer) return;
      const userVal = recallInput.value.trim().toLowerCase();
      const expected = card.answer.trim().toLowerCase();
      if (userVal === expected) {
        recallFeedback.style.color = "var(--good)";
        recallFeedback.textContent = `✅ Perfetto! ("${card.answer}")`;
      } else {
        recallFeedback.style.color = "var(--warn)";
        recallFeedback.textContent = `💡 Risposta attesa: "${card.answer}"`;
      }
    });

    recallInput.addEventListener("keydown", (e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        document.getElementById("checkRecallBtn").click();
      }
    });

    cardWrapper.addEventListener("click", (e) => {
      if (e.target.closest("button, input, select")) return;
      flipCard();
    });

    document.getElementById("flipBtn").addEventListener("click", flipCard);
    document.getElementById("nextBtn").addEventListener("click", nextCard);
    document.getElementById("prevBtn").addEventListener("click", prevCard);

    document.getElementById("btnAgain").addEventListener("click", () => rateCard("again"));
    document.getElementById("btnHard").addEventListener("click", () => rateCard("hard"));
    document.getElementById("btnGood").addEventListener("click", () => rateCard("good"));

    let touchStartX = 0;
    let touchStartY = 0;
    const touchStage = document.getElementById("touchStage");

    touchStage.addEventListener("touchstart", (e) => {
      touchStartX = e.changedTouches[0].screenX;
      touchStartY = e.changedTouches[0].screenY;
    }, { passive: true });

    touchStage.addEventListener("touchend", (e) => {
      const dx = e.changedTouches[0].screenX - touchStartX;
      const dy = e.changedTouches[0].screenY - touchStartY;
      if (Math.abs(dx) > 55 && Math.abs(dx) > Math.abs(dy) * 1.4) {
        if (dx < 0) nextCard();
        else prevCard();
      }
    }, { passive: true });

    document.addEventListener("keydown", (e) => {
      if (document.activeElement === recallInput || creatorModal.classList.contains("open")) return;
      if (e.code === "Space") { e.preventDefault(); flipCard(); }
      else if (e.code === "ArrowRight") nextCard();
      else if (e.code === "ArrowLeft") prevCard();
      else if (e.key === "1") rateCard("again");
      else if (e.key === "2") rateCard("hard");
      else if (e.key === "3") rateCard("good");
      else if (e.key.toLowerCase() === "s") toggleSaveCard(state.deck[state.index]);
    });

    // Init
    updateSavedBadge();
    updateSubFilterOptions();
    loadDeck();
  </script>
